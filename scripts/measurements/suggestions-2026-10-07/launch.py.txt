import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile

CHECKOUT = Path(os.environ['BRAIN_MEASUREMENT_CHECKOUT'])
SCRATCH = Path(os.environ['BRAIN_MEASUREMENT_SCRATCH'])
BUN = Path(os.environ['BRAIN_MEASUREMENT_BUN'])
CLAUDE = Path(os.environ['BRAIN_MEASUREMENT_CLAUDE'])
RTK = Path(os.environ['BRAIN_MEASUREMENT_RTK'])
SCRATCH.mkdir(mode=0o700, exist_ok=True)
for directory in ['config', 'home', 'pi', 'runs']:
    (SCRATCH / directory).mkdir(mode=0o700, exist_ok=True)

def launch(args, credentials=False, brain=None):
    invocation=Path(tempfile.mkdtemp(prefix='invocation-',dir=SCRATCH))
    for name in ['home','config','pi']:(invocation/name).mkdir(mode=0o700)
    env = {
        'PATH': f'{BUN.parent}:{RTK.parent}:/usr/bin:/bin',
        'HOME': str(invocation / 'home'),
        'CLAUDE_CONFIG_DIR': str(invocation / 'config'),
        'CLAUDE_CODE_PATH': str(CLAUDE),
        'PI_CODING_AGENT_DIR': str(invocation / 'pi'),
        'NODE_PATH': str(CHECKOUT / 'node_modules'),
        'TMPDIR': str(SCRATCH),
        'NO_PROXY': 'localhost,127.0.0.1,::1',
        'DO_NOT_TRACK': '1',
        **{key: os.environ[key] for key in ['BRAIN_MEASUREMENT_CHECKOUT','BRAIN_MEASUREMENT_SCRATCH','BRAIN_MEASUREMENT_BUN','BRAIN_MEASUREMENT_CLAUDE','BRAIN_MEASUREMENT_RTK']},
    }
    if brain:
        env['BRAIN_ROOT'] = str(brain)
    if os.environ.get('BRAIN_MEASURE_SUGGESTIONS_ARM') in ['rule', 'no-rule']:
        env['BRAIN_MEASURE_SUGGESTIONS_ARM'] = os.environ['BRAIN_MEASURE_SUGGESTIONS_ARM']
    if credentials and os.environ.get('BRAIN_MEASURE_BILLING_ROUTE')=='anthropic-api-key':
        if '--backend' not in args or args[args.index('--backend')+1]!='pi' or str(SCRATCH/'pi-api-capture-preload.ts') not in args:
            raise RuntimeError('API key route is restricted to the instrumented Pi backend')
        key=os.environ.get('ANTHROPIC_API_KEY')
        if not key:raise RuntimeError('Authorized Anthropic API key absent')
        env['ANTHROPIC_API_KEY']=key
        env['BRAIN_MEASURE_CELL']=os.environ.get('BRAIN_MEASURE_CELL','')
        env['BRAIN_MEASURE_BILLING_ROUTE']='anthropic-api-key'
    elif credentials:
        auth = json.loads((Path.home() / '.claude/.credentials.json').read_text())['claudeAiOauth']
        token = auth['accessToken']
        env['CLAUDE_CODE_OAUTH_TOKEN'] = token
        env['ANTHROPIC_OAUTH_TOKEN'] = token
    command = ['bwrap', '--unshare-all', '--share-net', '--die-with-parent', '--new-session', '--cap-drop', 'ALL']
    for source in ['/usr', '/bin', '/lib', '/lib64']:
        command += ['--ro-bind', source, source]
    for source in ['/etc/ssl', '/etc/resolv.conf', '/etc/hosts', '/etc/nsswitch.conf']:
        if Path(source).exists():
            command += ['--ro-bind', source, source]
    command += ['--ro-bind', str(CHECKOUT), str(CHECKOUT), '--ro-bind', str(BUN), str(BUN),
                '--ro-bind', str(CLAUDE), str(CLAUDE), '--ro-bind', str(RTK), str(RTK), '--bind', str(SCRATCH), str(SCRATCH),
                '--proc', '/proc', '--dev', '/dev', '--chdir', str(CHECKOUT), '--', str(BUN), *args]
    return subprocess.run(command, env=env).returncode

if sys.argv[1] == 'preflight':
    # Controlled, fictional sentinel outside the mounted scratch must be unreadable.
    sentinel = Path('/tmp/550-outside-sentinel.txt')
    sentinel.write_text('Odysseus: outside the admitted fixture.\n')
    code = '''import {existsSync,readFileSync,writeFileSync} from "node:fs";
const out="/tmp/550-outside-sentinel.txt";let denied=false;try{readFileSync(out)}catch{denied=true}
const scratch=process.env.TMPDIR;writeFileSync(`${scratch}/probe.txt`,"Odysseus: admitted scratch.");
const probe={outsideReadDenied:denied,scratchRead:readFileSync(`${scratch}/probe.txt`,"utf8"),sourcePresent:existsSync("scripts/measure-suggestions.ts"),claudeExecutable:existsSync(process.env.CLAUDE_CODE_PATH),personalHomeAbsent:!existsSync("__PERSONAL_CLAUDE_HOME__")};console.log(JSON.stringify(probe));if(!denied||!probe.sourcePresent||!probe.claudeExecutable||!probe.personalHomeAbsent)process.exit(1);'''
    code=code.replace('"__PERSONAL_CLAUDE_HOME__"',json.dumps(str(Path.home()/'.claude')))
    sys.exit(launch(['-e', code]))
elif sys.argv[1] == 'stage':
    for backend in ['claude', 'pi']:
        target = SCRATCH / f'{backend}-brain'
        if target.exists():
            raise RuntimeError('Existing staged fixture; refuse overwrite')
        shutil.copytree(CHECKOUT / 'packages/core/fixtures/corpus', target)
        status = launch(['packages/core/src/cli/brain.ts', 'index', '--force', '--json'], brain=target)
        if status:
            sys.exit(status)
elif sys.argv[1] == 'index':
    for backend in ['claude', 'pi']:
        status = launch(['packages/core/src/cli/brain.ts', 'index', '--force', '--json'], brain=SCRATCH / f'{backend}-brain')
        if status:
            sys.exit(status)
elif sys.argv[1] == 'run':
    sys.exit(launch(sys.argv[2:], credentials=True))
elif sys.argv[1] == 'network':
    sys.exit(launch(['-e', 'try { const response=await fetch("https://api.anthropic.com/v1/models", {signal:AbortSignal.timeout(10000)}); console.log(JSON.stringify({status:response.status})); } catch(error) { console.log(JSON.stringify({error:error.name,message:error.message})); process.exit(1); }']))
else:
    raise RuntimeError('Use preflight, stage, or run')
