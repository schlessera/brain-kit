"""Owned bin wrapper and readonly workspace source alias; no copied/shared dependencies."""
from pathlib import Path
import os,sys
CHECKOUT=Path(os.environ['BRAIN_MEASUREMENT_CHECKOUT'])
SCRATCH=Path(os.environ['BRAIN_MEASUREMENT_SCRATCH'])
def wrapper_bytes():
    template=(SCRATCH/'fixture-brain-wrapper.sh').read_text()
    for key in ['BRAIN_MEASUREMENT_CHECKOUT','BRAIN_MEASUREMENT_RTK','BRAIN_MEASUREMENT_BUN']:
        template=template.replace('${'+key+':?}',os.environ[key].replace('\\','\\\\').replace('"','\\"').replace('$','\\$').replace('`','\\`'))
    return template.encode()
def stage_runtime(brain):
    brain=Path(brain)
    if not brain.is_dir() or brain.is_symlink():raise RuntimeError('Expected owned fixture directory')
    bin_dir=brain/'node_modules/.bin';bin_dir.mkdir(parents=True,exist_ok=True)
    alias_dir=brain/'node_modules/@schlessera';alias_dir.mkdir(parents=True,exist_ok=True)
    target=alias_dir/'brain'
    if target.exists() or target.is_symlink():raise RuntimeError('Refuse existing runtime alias')
    target.symlink_to(CHECKOUT/'packages/core',target_is_directory=True)
    wrapper=bin_dir/'brain';wrapper.write_bytes(wrapper_bytes());wrapper.chmod(0o755)
def validate_runtime(brain):
    brain=Path(brain);wrapper=brain/'node_modules/.bin/brain';alias=brain/'node_modules/@schlessera/brain'
    if wrapper.is_symlink() or wrapper.read_bytes()!=wrapper_bytes() or not wrapper.stat().st_mode & 0o111:raise RuntimeError('Fixture bin wrapper drift')
    if not alias.is_symlink() or alias.readlink()!=CHECKOUT/'packages/core':raise RuntimeError('Readonly workspace source alias drift')
if __name__=='__main__':stage_runtime(sys.argv[1]);validate_runtime(sys.argv[1])
