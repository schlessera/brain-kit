import {plugin} from "bun";
import {readFileSync} from "node:fs";
const helper=`${process.env.TMPDIR}/query-capture.ts`;
plugin({name:"measure-550-raw-results",setup(build){
  build.onLoad({filter:/\/(measure-show-block\.ts|ui-backend-claude\/src\/turn-runner\.ts)$/},({path})=>{
    let source=readFileSync(path,"utf8");
    if(path.endsWith("/measure-show-block.ts")){
      const anchor="const { createSdkMcpServer, query } = (await import(SDK_ENTRY))";
      if(source.split(anchor).length!==2)throw new Error("SDK query anchor drifted");
      source=source.replace(anchor,"const { createSdkMcpServer, query: uncapturedQuery } = (await import(SDK_ENTRY))");
      const insertion="const BRAIN_PATH = stageBrain();";
      if(source.split(insertion).length!==2)throw new Error("SDK wrapper insertion anchor drifted");
      source=source.replace(insertion,`const query = (parameters: Parameters<typeof uncapturedQuery>[0]) => captureQuery(uncapturedQuery, {...parameters, options: {...parameters.options, cwd: stageBrain()}});\n${insertion}`);
    }else{
      const anchor="const queryFn = options.backend.queryFn ?? query;";
      if(source.split(anchor).length!==2)throw new Error("Backend query anchor drifted");
      source=source.replace(anchor,"const uncapturedQuery = options.backend.queryFn ?? query;\n  const queryFn = (parameters: Parameters<typeof query>[0]) => captureQuery(uncapturedQuery, parameters);");
    }
    return {contents:`import {captureQuery} from ${JSON.stringify(helper)};\n${source}`,loader:"ts"};
  });
}});
